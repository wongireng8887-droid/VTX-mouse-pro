# Update VTX Mouse Pro lewat Termux (Fase 2a)

Folder kerja: ~/storage/shared/VTXMousePro   (repo git ada di subfolder "repo")

ZIP=~/storage/downloads/VTXMousePro_fase2a.zip
[ -f "$ZIP" ] || ZIP=~/storage/shared/VTXMousePro/VTXMousePro_fase2a.zip
cd ~/storage/shared/VTXMousePro && \
ls -l "$ZIP" && \
rm -rf _unzip && mkdir _unzip && \
unzip -o "$ZIP" -d _unzip && \
test -f _unzip/VTXMousePro/app/build.gradle.kts && \
cd repo && \
rm -rf app && \
cp -r ../_unzip/VTXMousePro/. . && \
test -f app/build.gradle.kts && \
git add -A && \
git status --short | head -20 && \
git commit -m "Fase 2a: koneksi Shizuku otomatis + tes sentuhan" && \
git push

(Cek hasil git status: harus M/A, bukan banyak D.)
Setelah push: GitHub -> Actions -> tunggu hijau -> Artifacts -> VTX-Mouse-Pro-apk.
